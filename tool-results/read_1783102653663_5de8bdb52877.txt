    56→    56→
    57→    57→export function BasketPage({
    58→    58→  products,
    59→    59→  vendors,
    60→    60→  selection,
    61→    61→  setSelection,
    62→    62→  onCompare,
    63→    63→  onChooseVendor,
    64→    64→  onNavigate,
    65→    65→  activeCategory,
    66→    66→  onClearCategory,
    67→    67→}: {
    68→    68→  products: Product[];
    69→    69→  vendors: Vendor[];
    70→    70→  selection: Selection;
    71→    71→  setSelection: (s: Selection) => void;
    72→    72→  onCompare: () => Promise<CompareResult | null>;
    73→    73→  onChooseVendor: (vendorId: string, items: CartItem[]) => void;
    74→    74→  onNavigate: (p: PageId) => void;
    75→    75→  activeCategory: string | null;
    76→    76→  onClearCategory: () => void;
    77→    77→}) {
    78→    78→  const [loading, setLoading] = useState(false);
    79→    79→  const [result, setResult] = useState<CompareResult | null>(null);
    80→    80→  const [open, setOpen] = useState(false);
    81→    81→  const [error, setError] = useState<string | null>(null);
    82→    82→  const [budget, setBudget] = useState<string>("");
    83→    83→
    84→    84→  // Derive all unique categories from the product list

    89→    89→

    91→    91→  const displayProducts = useMemo(() => {

    93→    93→    return products.filter((p) => p.category === activeCategory);
    94→    94→  }, [products, activeCategory]);
    95→    95→
    96→    96→  // Products already in the basket (for upsell suggestions)
    97→    97→  const selectedIds = useMemo(
    98→    98→    () => new Set(Object.keys(selection).filter((id) => selection[id] > 0)),
    99→    99→    [selection]

   101→   101→
   102→   102→  // Upsell items: products in the same category NOT yet in the basket
   103→   103→  const upsellProducts = useMemo(() => {
   104→   104→    if (!activeCategory) return [];
   105→   105→    return products.filter(
   106→   106→      (p) => p.category === activeCategory && !selectedIds.has(p.id)
   107→   107→    );
   108→   108→  }, [products, activeCategory, selectedIds]);
   109→   109→
   110→   110→  const totalQty = Object.values(selection).reduce((a, b) => a + b, 0);
   111→   111→  const hasItems = totalQty > 0;
   112→   112→
   113→   113→  const estimatedMin = products.reduce((s, p) => {
   114→   114→    const q = selection[p.id] || 0;
   115→   115→    return s + p.bestPrice * q;
   116→   116→  }, 0);
   117→   117→  const budgetNum = Number(budget) || 0;
   118→   118→  const overBudget = budgetNum > 0 && estimatedMin > budgetNum;
   119→   119→  const budgetPct =
   120→   120→    budgetNum > 0 ? Math.min(100, Math.round((estimatedMin / budgetNum) * 100)) : 0;
   121→   121→
   122→   122→  const meta = activeCategory ? CATEGORY_META[activeCategory] : null;
   123→   123→
   124→   124→  const handleCompare = async () => {
   125→   125→    setLoading(true);
   126→   126→    setError(null);
   127→   127→    try {
   128→   128→      const res = await onCompare();
   129→   129→      if (!res) {
   130→   130→        setError("Could not compare prices. Please try again.");
   131→   131→        return;
   132→   132→      }
   133→   133→      setResult(res);
   134→   134→      setOpen(true);
   135→   135→    } catch {
   136→   136→      setError("Something went wrong. Please try again.");
   137→   137→    } finally {
   138→   138→      setLoading(false);
   139→   139→    }
   140→   140→  };
   141→   141→
   142→   142→  const handleChoose = (vendorId: string, lineItems: CompareResult["bestSingle"]["lineItems"]) => {
   143→   143→    const items: CartItem[] = lineItems.map((li) => ({
   144→   144→      productId: li.productId,
   145→   145→      name: li.name,
   146→   146→      unit: li.unit,
   147→   147→      emoji: li.emoji,
   148→   148→      qty: li.qty,
   149→   149→      unitPrice: li.price,
   150→   150→    }));
   151→   151→    onChooseVendor(vendorId, items);
   152→   152→    setOpen(false);
   153→   153→    onNavigate("cart");
   154→   154→  };
   155→   155→
   156→   156→  const handleCategoryClick = (cat: string | null) => {
   157→   157→    if (cat === activeCategory) {
   158→   158→      onClearCategory();
   159→   159→    } else {
   160→   160→      onNavigate("basket", cat);
   161→   161→    }
   162→   162→  };
   163→   163→
   164→   164→  return (
   165→   165→    <div className="space-y-5">
   166→   166→      {/* Header */}
   167→   167→      <div className="space-y-2">
   168→   168→        {activeCategory ? (
   169→   169→          <div className="flex items-center gap-3">
   170→   170→            <button
   171→   171→              onClick={() => onClearCategory()}
   172→   172→              className="grid h-9 w-9 shrink-0 place-items-center rounded-full border border-border bg-card transition-colors hover:bg-muted"
   173→   173→              aria-label="Back to all products"
   174→   174→            >
   175→   175→              <ArrowLeft size={16} />
   176→   176→            </button>
   177→   177→            <div className="flex items-center gap-2">
   178→   178→              <span className="text-2xl">{meta?.emoji}</span>
   179→   179→              <h1 className="text-2xl font-extrabold">{activeCategory}</h1>
   180→   180→              <Badge className={cn("text-[10px]", meta?.color)}>
   181→   181→                {displayProducts.length} items
   182→   182→              </Badge>
   183→   183→            </div>
   184→   184→          </div>
   185→   185→        ) : (
   186→   186→          <div className="flex items-center gap-2">
   187→   187→            <ShoppingBasket className="text-brand" size={24} />
   188→   188→            <h1 className="text-2xl font-extrabold">Smart Basket</h1>
   189→   189→          </div>
   190→   190→        )}
   191→   191→        <p className="text-sm text-muted-foreground">
   192→   192→          {activeCategory
   193→   193→            ? `Browse ${activeCategory.toLowerCase()} items — add to your basket, compare vendors, and pick the best deal.`
   194→   194→            : "Build your list once — WeBizzle checks every vendor and finds the cheapest full basket."}
   195→   195→        </p>
   196→   196→      </div>
   197→   197→
   198→   198→      {/* Category pills (only when no active category) */}
   199→   199→      {!activeCategory && (
   200→   200→        <div className="flex gap-2 overflow-x-auto pb-1 scrollbar-none">
   201→   201→          {allCategories.map((cat) => {
   202→   202→            const cm = CATEGORY_META[cat];
   203→   203→            const count = products.filter((p) => p.category === cat).length;
   204→   204→            if (count === 0) return null;
   205→   205→            return (
   206→   206→              <button
   207→   207→                key={cat}
   208→   208→                onClick={() => handleCategoryClick(cat)}
   209→   209→                className="inline-flex shrink-0 items-center gap-1.5 rounded-full border border-border bg-card px-3 py-1.5 text-xs font-medium transition-all hover:border-brand hover:bg-brand-light/40"
   210→   210→              >
   211→   211→                <span>{cm?.emoji}</span>
   212→   212→                {cat}
   213→   213→                <span className="text-muted-foreground">({count})</span>
   214→   214→              </button>
   215→   215→            );
   216→   216→          })}
   217→   217→        </div>
   218→   218→      )}
   219→   219→
   220→   220→      {/* Main product builder */}
   221→   221→      <Card className="divide-y divide-border overflow-hidden">
   222→   222→        {displayProducts.length === 0 ? (
   223→   223→          <div className="flex flex-col items-center gap-2 p-10 text-center">
   224→   224→            <span className="text-4xl">{meta?.emoji ?? "\u{1F6D2}"}</span>
   225→   225→            <p className="text-sm font-medium text-muted-foreground">
   226→   226→              No products in this category yet
   227→   227→            </p>
   228→   228→            <Button
   229→   229→              variant="outline"
   230→   230→              size="sm"
   231→   231→              onClick={() => onClearCategory()}
   232→   232→            >
   233→   233→              Browse all categories
   234→   234→            </Button>
   235→   235→          </div>
   236→   236→        ) : (
   237→   237→          displayProducts.map((p) => {
   238→   238→            const qty = selection[p.id] || 0;
   239→   239→            const active = qty > 0;
   240→   240→            return (
   241→   241→              <div
   242→   242→                key={p.id}
   243→   243→                className={cn(
   244→   244→                  "flex items-center gap-3 p-3 transition-colors sm:p-4",
   245→   245→                  active && "bg-brand-light/40"
   246→   246→                )}
   247→   247→              >
   248→   248→                <div
   249→   249→                  className={cn(
   250→   250→                    "grid h-12 w-12 shrink-0 place-items-center rounded-xl text-2xl",
   251→   251→                    active ? "bg-brand-light" : "bg-muted"
   252→   252→                  )}
   253→   253→                >
   254→   254→                  {p.emoji}
   255→   255→                </div>
   256→   256→                <div className="min-w-0 flex-1">
   257→   257→                  <div className="font-semibold leading-tight">{p.name}</div>
   258→   258→                  <div className="text-xs text-muted-foreground">
   259→   259→                    {p.unit} · from {KES(p.bestPrice)}
   260→   260→                  </div>
   261→   261→                </div>
   262→   262→                <Stepper
   263→   263→                  qty={qty}
   264→   264→                  onDec={() => {
   265→   265→                    const next = { ...selection };
   266→   266→                    if (qty <= 1) delete next[p.id];
   267→   267→                    else next[p.id] = qty - 1;
   268→   268→                    setSelection(next);
   269→   269→                  }}
   270→   270→                  onInc={() =>
   271→   271→                    setSelection({ ...selection, [p.id]: qty + 1 })
   272→   272→                  }
   273→   273→                />
   274→   274→              </div>
   275→   275→            );
   276→   276→          })
   277→   277→        )}
   278→   278→      </Card>
   279→   279→
   280→   280→      {/* UPSELL SUGGESTIONS (when category is active and items are in basket) */}
   281→   281→      {activeCategory && upsellProducts.length > 0 && hasItems && (
   282→   282→        <div className="space-y-3">
   283→   283→          <div className="flex items-center gap-2">
   284→   284→            <Sparkles className="h-4 w-4 text-gold" />
   285→   285→            <h3 className="text-sm font-bold">
   286→   286→              You might also need ({upsellProducts.length} more {activeCategory.toLowerCase()} items)
   287→   287→            </h3>
   288→   288→          </div>
   289→   289→          <div className="grid grid-cols-2 gap-2 sm:grid-cols-3">
   290→   290→            {upsellProducts.slice(0, 6).map((p) => (
   291→   291→              <button
   292→   292→                key={p.id}
   293→   293→                onClick={() =>
   294→   294→                  setSelection({ ...selection, [p.id]: (selection[p.id] || 0) + 1 })
   295→   295→                }
   296→   296→                className="group flex items-center gap-2 rounded-xl border border-border bg-card p-2.5 text-left transition-all hover:border-brand hover:shadow-sm"
   297→   297→              >
   298→   298→                <span className="grid h-9 w-9 shrink-0 place-items-center rounded-lg bg-muted text-lg">
   299→   299→                  {p.emoji}
   300→   300→                </span>
   301→   301→                <div className="min-w-0 flex-1">
   302→   302→                  <div className="truncate text-xs font-semibold group-hover:text-brand">
   303→   303→                    {p.name}
   304→   304→                  </div>
   305→   305→                  <div className="text-[10px] text-muted-foreground">
   306→   306→                    {KES(p.bestPrice)}
   307→   307→                  </div>
   308→   308→                </div>
   309→   309→                <Plus
   310→   310→                  size={14}
   311→   311→                  className="shrink-0 text-muted-foreground group-hover:text-brand"
   312→   312→                />
   313→   313→              </button>
   314→   314→            ))}
   315→   315→          </div>
   316→   316→        </div>
   317→   317→      )}
   318→   318→
   319→   319→      {/* Already-in-basket quick view (when on category page, show what's selected) */}
   320→   320→      {activeCategory && hasItems && (
   321→   321→        <div className="space-y-2">
   322→   322→          <div className="flex items-center gap-2">
   323→   323→            <ShoppingBasket className="h-4 w-4 text-brand" />
   324→   324→            <h3 className="text-sm font-bold">In your basket ({totalQty} items)</h3>
   325→   325→          </div>
   326→   326→          <div className="flex flex-wrap gap-1.5">
   327→   327→            {Object.entries(selection)
   328→   328→              .filter(([, q]) => q > 0)
   329→   329→              .map(([pid, qty]) => {
   330→   330→                const p = products.find((pr) => pr.id === pid);
   331→   331→                if (!p) return null;
   332→   332→                return (
   333→   333→                  <Badge
   334→   334→                    key={pid}
   335→   335→                    variant="secondary"
   336→   336→                    className="gap-1 bg-brand-light/60 px-2 py-1 text-xs"
   337→   337→                  >
   338→   338→                    <span>{p.emoji}</span>
   339→   339→                    {p.name} x{qty}
   340→   340→                    <button
   341→   341→                      onClick={() => {
   342→   342→                        const next = { ...selection };
   343→   343→                        if (qty <= 1) delete next[pid];
   344→   344→                        else next[pid] = qty - 1;
   345→   345→                        setSelection(next);
   346→   346→                      }}
   347→   347→                      className="ml-0.5 rounded-full p-0.5 hover:bg-brand/20"
   348→   348→                      aria-label={`Remove ${p.name}`}
   349→   349→                    >
   350→   350→                      <X size={10} />
   351→   351→                    </button>
   352→   352→                  </Badge>
   353→   353→                );
   354→   354→              })}
   355→   355→          </div>
   356→   356→        </div>
   357→   357→      )}
   358→   358→
   359→   359→      {/* Summary + Compare CTA */}
   360→   360→      <div className="sticky bottom-16 z-30 md:bottom-4">
   361→   361→        <Card className="border-brand/30 bg-card/95 p-4 shadow-lg backdrop-blur">
   362→   362→          {/* Budget guardrail */}
   363→   363→          <div className="mb-3 flex items-center gap-2">
   364→   364→            <span className="inline-flex items-center gap-1 text-xs font-medium text-muted-foreground">
   365→   365→              <Wallet size={13} /> Budget
   366→   366→            </span>
   367→   367→            <div className="relative flex-1">
   368→   368→              <span className="pointer-events-none absolute left-2.5 top-1/2 -translate-y-1/2 text-xs text-muted-foreground">
   369→   369→                KES
   370→   370→              </span>
   371→   371→              <input
   372→   372→                type="number"
   373→   373→                inputMode="numeric"
   374→   374→                min={0}
   375→   375→                placeholder="Set a max spend (optional)"
   376→   376→                value={budget}
   377→   377→                onChange={(e) => setBudget(e.target.value)}
   378→   378→                className="h-8 w-full rounded-lg border border-border bg-background pl-9 pr-2 text-xs outline-none focus:border-brand focus:ring-1 focus:ring-brand"
   379→   379→              />
   380→   380→            </div>
   381→   381→          </div>
   382→   382→
   383→   383→          {budgetNum > 0 && estimatedMin > 0 && (
   384→   384→            <div className="mb-3">
   385→   385→              <div className="flex items-center justify-between text-[11px]">
   386→   386→                <span className={overBudget ? "font-semibold text-destructive" : "text-muted-foreground"}>
   387→   387→                  Est. min {KES(estimatedMin)}
   388→   388→                </span>
   389→   389→                <span className="text-muted-foreground">
   390→   390→                  Budget {KES(budgetNum)}
   391→   391→                </span>
   392→   392→              </div>
   393→   393→              <div className="mt-1 h-2 overflow-hidden rounded-full bg-muted">
   394→   394→                <div
   395→   395→                  className={
   396→   396→                    "h-full rounded-full transition-all " +
   397→   397→                    (overBudget
   398→   398→                      ? "bg-destructive"
   399→   399→                      : "bg-gradient-to-r from-brand to-brand-dark")
   400→   400→                  }
   401→   401→                  style={{ width: `${budgetPct}%` }}
   402→   402→                />
   403→   403→              </div>
   404→   404→              {overBudget ? (
   405→   405→                <div className="mt-1 text-[11px] font-medium text-destructive">
   406→   406→                  Over budget by {KES(estimatedMin - budgetNum)} — compare to
   407→   407→                  find savings, or trim your list.
   408→   408→                </div>
   409→   409→              ) : (
   410→   410→                <div className="mt-1 text-[11px] text-muted-foreground">
   411→   411→                  On track — {KES(budgetNum - estimatedMin)} of headroom.
   412→   412→                </div>
   413→   413→              )}
   414→   414→            </div>
   415→   415→          )}
   416→   416→
   417→   417→          <div className="flex items-center justify-between gap-3">
   418→   418→            <div>
   419→   419→              <div className="text-xs text-muted-foreground">
   420→   420→                {Object.keys(selection).length} product
   421→   421→                {Object.keys(selection).length !== 1 ? "s" : ""} · {totalQty}{" "}
   422→   422→                item{totalQty !== 1 ? "s" : ""}
   423→   423→              </div>
   424→   424→              <div className="text-sm font-semibold text-foreground">
   425→   425→                {activeCategory
   426→   426→                  ? `Comparing across ${vendors.filter((v) => v.type === activeCategory).length || vendors.length} ${activeCategory.toLowerCase()} vendor${vendors.filter((v) => v.type === activeCategory).length !== 1 ? "s" : ""}+ more`
   427→   427→                  : `We'll compare ${Object.keys(selection).length || 0} product${Object.keys(selection).length !== 1 ? "s" : ""} across ${vendors.length} vendors`}
   428→   428→              </div>
   429→   429→            </div>
   430→   430→            <Button
   431→   431→              className="bg-brand text-white hover:bg-brand-dark"
   432→   432→              disabled={!hasItems || loading}
   433→   433→              onClick={handleCompare}
   434→   434→            >
   435→   435→              {loading ? (
   436→   436→                <Loader2 size={18} className="animate-spin" />
   437→   437→              ) : (
   438→   438→                <ShoppingBag size={18} />
   439→   439→              )}
   440→   440→              Compare ({totalQty})
   441→   441→            </Button>
   442→   442→          </div>
   443→   443→          {error && (
   444→   444→            <div className="mt-2 text-xs text-destructive">{error}</div>
   445→   445→          )}
   446→   446→        </Card>
   447→   447→      </div>
   448→   448→
   449→   449→      <CompareDialog
   450→   450→        open={open}
   451→   451→        onOpenChange={setOpen}
   452→   452→        result={result}
   453→   453→        onChoose={handleChoose}
   454→   454→        budget={budgetNum}
   455→   455→      />
   456→   456→    </div>
   457→   457→  );
   458→   458→}
   459→   459→
   460→   460→function Stepper({
   461→   461→  qty,
   462→   462→  onDec,
