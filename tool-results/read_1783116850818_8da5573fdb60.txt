    41→function getPortalToken(): string | null {
    42→  if (typeof window === "undefined") return null;
    43→  try { return sessionStorage.getItem(PORTAL_TOKEN_KEY); } catch { return null; }

    50→// ---------- helpers ----------
    51→function relativeTime(iso: string): string {
    52→  const then = new Date(iso).getTime();
    53→  if (Number.isNaN(then)) return "";
    54→  const diff = Date.now() - then;
    55→  if (diff < 0) return "just now";
    56→  const s = Math.floor(diff / 1000);
    57→  if (s < 45) return "just now";
    58→  const m = Math.floor(s / 60);
    59→  if (m < 60) return `${m}m ago`;
    60→  const h = Math.floor(m / 60);
    61→  if (h < 24) return `${h}h ago`;
    62→  const d = Math.floor(h / 24);
    63→  if (d < 7) return `${d}d ago`;
    64→  return new Date(iso).toLocaleDateString();
    65→}

    67→function nowHHMM(): string {
    68→  const d = new Date();
    69→  return `${String(d.getHours()).padStart(2, "0")}:${String(
    70→    d.getMinutes()
    71→  ).padStart(2, "0")}`;
    72→}

    74→// True if `now` falls inside the [start, end) duty window. Handles overnight
    75→// windows (e.g. 22:00 → 06:00). Empty schedule ⇒ treat as always on duty.
    76→function isWithinDuty(start: string, end: string, now = nowHHMM()): boolean {
    77→  if (!start || !end) return true;
    78→  if (start === end) return false;
    79→  if (start < end) return now >= start && now < end;
    80→  return now >= start || now < end; // overnight
    81→}

   590→              <div className="flex items-center gap-4">
   591→                <div className="flex size-14 items-center justify-center rounded-xl bg-brand-light text-3xl">
   592→                  {vendor?.emoji ?? "🏪"}
   593→                </div>
   594→                <div>
   595→                  <div className="flex items-center gap-2">
   596→                    <h2 className="text-lg font-bold text-foreground">
   597→                      {portal?.shopName ?? vendor?.name ?? "Vendor"}
   598→                    </h2>
   599→                    <Badge
   600→                      variant={isOnline ? "default" : "secondary"}
   601→                      className={cn(
   602→                        "gap-1.5",
   603→                        isOnline
   604→                          ? "bg-brand text-primary-foreground"
   605→                          : "bg-muted text-muted-foreground"
   606→                      )}
   607→                    >
   608→                      <OnlineDot online={isOnline} />
   609→                      {isOnline ? "Online" : "Offline"}
   610→                    </Badge>
   611→                  </div>
   612→                  <div className="mt-1 flex flex-wrap items-center gap-x-3 gap-y-1 text-xs text-muted-foreground">
   613→                    {vendor?.type && (
   614→                      <span className="inline-flex items-center gap-1">
   615→                        <Store size={12} /> {vendor.type}
   616→                      </span>
   617→                    )}
   618→                    {vendor?.location && (
   619→                      <span className="inline-flex items-center gap-1">
   620→                        <MapPin size={12} /> {vendor.location}
   621→                      </span>
   622→                    )}
   623→                    {typeof vendor?.rating === "number" && (
   624→                      <span className="inline-flex items-center gap-1">
   625→                        <Star size={12} className="fill-gold text-gold" />
   626→                        {vendor.rating.toFixed(1)}
   627→                      </span>
   628→                    )}
   629→                    {connected && (
   630→                      <span className="inline-flex items-center gap-1 text-brand">
   631→                        <span className="size-1.5 rounded-full bg-brand animate-pulse" />
   632→                        realtime live
   633→                      </span>
   634→                    )}
   635→                  </div>
   636→                </div>
   637→              </div>
   638→              <Button
   639→                onClick={toggleOnline}
   640→                disabled={patching}
   641→                className={cn(
   642→                  "h-11 px-6",
   643→                  isOnline
   644→                    ? "bg-muted text-foreground hover:bg-muted/80"
   645→                    : "bg-brand text-primary-foreground hover:bg-brand-dark"
   646→                )}
   647→              >
   648→                {patching ? (
   649→                  <Loader2 size={16} className="animate-spin" />
   650→                ) : (
   651→                  <Power size={16} />
   652→                )}
   653→                {isOnline ? "Go offline" : "Go online"}
   654→              </Button>
   655→            </div>
   656→          )}
   657→        </CardContent>
   658→      </Card>
   659→
   660→      {/* duty hours + phone */}
   661→      <Card>
   662→        <CardContent className="p-6">
   663→          <div className="mb-4 flex items-center justify-between gap-2">
   664→            <h3 className="flex items-center gap-2 text-sm font-semibold text-foreground">
   665→              <Clock size={16} className="text-brand" /> Duty hours
   666→            </h3>
   667→            <Badge
   668→              variant="outline"
   669→              className={cn(
   670→                onDuty
   671→                  ? "border-brand/30 text-brand"
   672→                  : "text-muted-foreground"
   673→              )}
   674→            >
   675→              {dutyStart && dutyEnd
   676→                ? onDuty
   677→                  ? "Currently on duty"
   678→                  : "Off duty"
   679→                : "No schedule set"}
   680→            </Badge>
   681→          </div>
   682→          <div className="grid grid-cols-1 gap-4 sm:grid-cols-2 lg:grid-cols-3">
   683→            <div className="space-y-1.5">
   684→              <Label htmlFor="duty-start" className="text-xs">
   685→                Start
   686→              </Label>
   687→              <Input
   688→                id="duty-start"
   689→                type="time"
   690→                value={dutyStart}
   691→                onChange={(e) => setDutyStart(e.target.value)}
   692→                onBlur={handleDutyBlur}
   693→                disabled={patching}
   694→              />
   695→            </div>
   696→            <div className="space-y-1.5">
   697→              <Label htmlFor="duty-end" className="text-xs">
   698→                End
   699→              </Label>
   700→              <Input
   701→                id="duty-end"
   702→                type="time"
   703→                value={dutyEnd}
   704→                onChange={(e) => setDutyEnd(e.target.value)}
   705→                onBlur={handleDutyBlur}
   706→                disabled={patching}
   707→              />
   708→            </div>
   709→            <div className="space-y-1.5">
   710→              <Label htmlFor="phone" className="text-xs">
   711→                Shop phone
   712→              </Label>
   713→              <Input
   714→                id="phone"
   715→                inputMode="tel"
   716→                placeholder="0712 345 678"
   717→                value={phoneInput}
   718→                onChange={(e) => setPhoneInput(e.target.value)}
   719→                onBlur={handlePhoneBlur}
   720→                disabled={patching}
   721→              />
   722→            </div>
   723→          </div>
   724→          {isOnline && !onDuty && dutyStart && dutyEnd && (
   725→            <p className="mt-3 text-xs text-gold">
   726→              You&apos;re online outside your duty window ({dutyStart}–
   727→              {dutyEnd}). That&apos;s fine — orders will still come through.
   728→            </p>
   729→          )}
   730→        </CardContent>
   731→      </Card>
   732→
   733→      {/* presence strip */}
   734→      <Card>
   735→        <CardContent className="p-4">
   736→          <div className="flex flex-wrap items-center gap-2">
   737→            <span className="inline-flex items-center gap-1.5 text-xs font-semibold text-muted-foreground">
   738→              <Users size={14} /> Online now
   739→            </span>
   740→            {otherPresence.length === 0 ? (
   741→              <span className="text-xs text-muted-foreground">
   742→                No one else online right now.
   743→              </span>
   744→            ) : (
   745→              otherPresence.map((p: PresenceEntry) => (
   746→                <span
   747→                  key={`${p.kind}:${p.id}`}
   748→                  className={cn(
   749→                    "inline-flex items-center gap-1.5 rounded-full border px-2.5 py-1 text-xs font-medium",
   750→                    p.kind === "vendor" &&
   751→                      "border-brand/30 bg-brand-light text-brand",
   752→                    p.kind === "rider" &&
   753→                      "border-gold/40 bg-gold/10 text-gold-dark",
   754→                    p.kind === "admin" &&
   755→                      "border-mpesa/40 bg-mpesa/10 text-mpesa"
   756→                  )}
   757→                >
   758→                  <span className="size-1.5 rounded-full bg-current opacity-70" />
   759→                  <span className="capitalize opacity-70">{p.kind}</span>
   760→                  {p.name}
   761→                </span>
   762→              ))
   763→            )}
   764→          </div>
   765→        </CardContent>
   766→      </Card>
   767→
   768→      {/* main grid: notifications | chat */}
   769→      <div className="grid grid-cols-1 gap-5 lg:grid-cols-2">
   770→        {/* notifications */}
   771→        <Card>
   772→          <CardContent className="p-6">
   773→            <div className="mb-3 flex items-center justify-between gap-2">
   774→              <h3 className="flex items-center gap-2 text-sm font-semibold text-foreground">
   775→                <Bell size={16} className="text-brand" /> Order notifications
   776→                {unreadCount > 0 && (
   777→                  <Badge className="bg-brand text-primary-foreground">
   778→                    {unreadCount} new
   779→                  </Badge>
   780→                )}
   781→              </h3>
   782→              {unreadCount > 0 && (
   783→                <Button
   784→                  variant="ghost"
   785→                  size="sm"
   786→                  className="h-7 text-xs text-muted-foreground"
   787→                  onClick={markAllRead}
   788→                >
   789→                  <CheckCheck size={14} /> Mark all read
   790→                </Button>
   791→              )}
   792→            </div>
   793→            {notifError ? (
   794→              <p className="py-6 text-center text-sm text-destructive">
   795→                {notifError}
   796→              </p>
   797→            ) : loadingNotifs && notifications.length === 0 ? (
   798→              <div className="flex items-center justify-center gap-2 py-8 text-muted-foreground">
   799→                <Loader2 size={16} className="animate-spin" /> Loading…
   800→              </div>
   801→            ) : notifications.length === 0 ? (
   802→              <div className="flex flex-col items-center gap-2 py-10 text-center">
   803→                <BellOff className="text-muted-foreground" size={28} />
   804→                <p className="text-sm text-muted-foreground">
   805→                  No notifications yet. New orders will appear here instantly.
   806→                </p>
   807→              </div>
   808→            ) : (
   809→              <div className="max-h-96 space-y-2 overflow-y-auto wb-scroll pr-1">
   810→                {notifications.map((n) => (
   811→                  <button
   812→                    key={n.id}
   813→                    type="button"
   814→                    onClick={() => !n.read && markRead(n.id)}
   815→                    className={cn(
   816→                      "w-full rounded-lg border p-3 text-left transition-colors",
   817→                      n.read
   818→                        ? "border-border bg-card hover:bg-accent/40"
   819→                        : "border-brand/30 bg-brand-light hover:bg-brand-light/70"
   820→                    )}
   821→                  >
   822→                    <div className="flex items-start justify-between gap-2">
   823→                      <span className="text-sm font-semibold text-foreground">
   824→                        {n.title}
   825→                      </span>
   826→                      <span className="shrink-0 text-[11px] text-muted-foreground">
   827→                        {relativeTime(n.createdAt)}
   828→                      </span>
   829→                    </div>
   830→                    <p className="mt-0.5 text-xs text-muted-foreground">
   831→                      {n.body}
   832→                    </p>
   833→                    <div className="mt-1.5 flex items-center gap-2">
   834→                      {n.type === "NEW_ORDER" && (
   835→                        <Badge className="bg-brand text-primary-foreground">
   836→                          New order
   837→                        </Badge>
   838→                      )}
   839→                      {n.orderId && (
   840→                        <span className="text-[11px] text-muted-foreground">
   841→                          #{n.orderId}
   842→                        </span>
   843→                      )}
   844→                      {!n.read && (
   845→                        <span className="ml-auto text-[11px] font-medium text-brand">
   846→                          Tap to mark read
   847→                        </span>
   848→                      )}
   849→                    </div>
   850→                  </button>
   851→                ))}
   852→              </div>
   853→            )}
   854→          </CardContent>
   855→        </Card>
   856→
   857→        {/* chat */}
   858→        <Card>
   859→          <CardContent className="p-6">
   860→            <div className="mb-3 flex items-center justify-between gap-2">
   861→              <h3 className="flex items-center gap-2 text-sm font-semibold text-foreground">
   862→                <MessageSquare size={16} className="text-brand" /> Support chat
   863→              </h3>
   864→              <span className="text-[11px] text-muted-foreground">
   865→                {socket ? "Live" : "Connect to chat live"}
   866→              </span>
   867→            </div>
   868→            <p className="mb-3 text-xs text-muted-foreground">
   869→              Channel{" "}
   870→              <code className="rounded bg-muted px-1 py-0.5 text-[11px]">
   871→                {channel}
   872→              </code>{" "}
   873→              — message WeBizzle admin support.
   874→            </p>
   875→
   876→            {chatError ? (
   877→              <p className="py-6 text-center text-sm text-destructive">
   878→                {chatError}
   879→              </p>
   880→            ) : loadingChat && messages.length === 0 ? (
   881→              <div className="flex items-center justify-center gap-2 py-8 text-muted-foreground">
   882→                <Loader2 size={16} className="animate-spin" /> Loading chat…
   883→              </div>
   884→            ) : (
   885→              <div
   886→                ref={chatListRef}
   887→                className="max-h-96 min-h-[14rem] space-y-2 overflow-y-auto wb-scroll rounded-lg bg-muted/40 p-3 pr-1"
   888→              >
   889→                {messages.length === 0 ? (
   890→                  <p className="py-8 text-center text-xs text-muted-foreground">
   891→                    No messages yet. Say hello to admin support 👋
   892→                  </p>
   893→                ) : (
   894→                  messages.map((m) => {
   895→                    const own =
   896→                      m.senderType === "VENDOR" && m.senderId === vendorId;
   897→                    return (
   898→                      <div
   899→                        key={m.id}
   900→                        className={cn(
   901→                          "flex flex-col",
   902→                          own ? "items-end" : "items-start"
   903→                        )}
   904→                      >
   905→                        {!own && (
   906→                          <span className="mb-0.5 px-1 text-[11px] font-medium text-muted-foreground">
   907→                            {m.senderName || "Admin"}
   908→                          </span>
   909→                        )}
   910→                        <div
   911→                          className={cn(
   912→                            "max-w-[85%] rounded-2xl px-3 py-2 text-sm",
   913→                            own
   914→                              ? "bg-brand text-primary-foreground rounded-br-sm"
   915→                              : "bg-card border text-card-foreground rounded-bl-sm"
   916→                          )}
   917→                        >
   918→                          {m.body}
   919→                        </div>
   920→                        <span className="mt-0.5 px-1 text-[10px] text-muted-foreground">
   921→                          {relativeTime(m.createdAt)}
   922→                        </span>
   923→                      </div>
   924→                    );
   925→                  })
   926→                )}
   927→              </div>
   928→            )}
   929→
   930→            <form
   931→              className="mt-3 flex items-center gap-2"
   932→              onSubmit={(e) => {
   933→                e.preventDefault();
   934→                sendChat();
   935→              }}
   936→            >
   937→              <Input
   938→                placeholder="Type a message…"
   939→                value={chatDraft}
   940→                onChange={(e) => setChatDraft(e.target.value)}
   941→                disabled={sending || !portal}
   942→                maxLength={1000}
   943→              />
   944→              <Button
   945→                type="submit"
   946→                size="icon"
   947→                className="bg-brand text-primary-foreground hover:bg-brand-dark"
   948→                disabled={sending || !chatDraft.trim() || !portal}
   949→                aria-label="Send message"
   950→              >
   951→                {sending ? (
   952→                  <Loader2 size={16} className="animate-spin" />
   953→                ) : (
   954→                  <Send size={16} />
   955→                )}
   956→              </Button>
   957→            </form>
   958→            {!socket && (
   959→              <p className="mt-2 flex items-center gap-1 text-[11px] text-muted-foreground">
   960→                <Phone size={11} /> Go online to receive live replies from
   961→                admin support.
   962→              </p>
   963→            )}
   964→          </CardContent>
   965→        </Card>
   966→      </div>
   967→    </div>
   968→  );
   969→}
