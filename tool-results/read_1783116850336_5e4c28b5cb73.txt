    59→function relativeTime(iso: string): string {
    60→  const then = new Date(iso).getTime();
    61→  if (Number.isNaN(then)) return "";
    62→  const diff = Math.max(0, Date.now() - then);
    63→  const s = Math.floor(diff / 1000);
    64→  if (s < 60) return "just now";
    65→  const m = Math.floor(s / 60);
    66→  if (m < 60) return `${m}m ago`;
    67→  const h = Math.floor(m / 60);
    68→  if (h < 24) return `${h}h ago`;
    69→  const d = Math.floor(h / 24);
    70→  return `${d}d ago`;
    71→}

    73→function statusPillClass(status: string): string {
    74→  switch ((status || "").toUpperCase()) {
    75→    case "DELIVERED":

    77→    case "DISPATCHED":

    81→    case "PLACED":

    85→  }

    88→// Normalise a raw socket "notification" payload (which may be the AppNotification
    89→// shape OR the leaner {type, orderId, customerName, total, at} shape the realtime
    90→// service emits) into an AppNotification-shaped object.
    91→function normalizeSocketNotif(
    92→  raw: unknown,
    93→  riderKey: string
    94→): AppNotification | null {
    95→  if (!raw || typeof raw !== "object") return null;
    96→  const n = raw as Record<string, unknown>;
    97→  const at = (n.at as string) || (n.createdAt as string) || new Date().toISOString();
    98→  const type = (n.type as string) || "SYSTEM";
    99→  const id =
   100→    (n.id as string) || `socket-${at}-${type}-${(n.orderId as string) || ""}`;
   101→  const title =
   102→    (n.title as string) ||
   103→    (type === "NEW_ORDER" ? "New order 🛒" : type === "DISPATCH" ? "Dispatch alert 🛵" : "New alert");
   104→  const body =
   105→    (n.body as string) ||
   106→    (n.customerName && typeof n.total === "number"
   107→      ? `${n.customerName} · KES ${Number(n.total).toLocaleString("en-KE")}`
   108→      : "");
   109→  return {
   110→    id,
   111→    recipientType: "RIDER",
   112→    recipientId: riderKey,
   113→    type,
   114→    title,
   115→    body,
   116→    read: false,
   117→    orderId: (n.orderId as string) || null,
   118→    createdAt: at,
   119→  };
   120→}
