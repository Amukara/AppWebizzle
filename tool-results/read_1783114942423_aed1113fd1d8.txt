    58→    58→function relativeTime(iso: string): string {
    59→    59→  const then = new Date(iso).getTime();
    60→    60→  if (Number.isNaN(then)) return "";
    61→    61→  const diff = Math.max(0, Date.now() - then);
    62→    62→  const s = Math.floor(diff / 1000);
    63→    63→  if (s < 60) return "just now";
    64→    64→  const m = Math.floor(s / 60);
    65→    65→  if (m < 60) return `${m}m ago`;
    66→    66→  const h = Math.floor(m / 60);
    67→    67→  if (h < 24) return `${h}h ago`;
    68→    68→  const d = Math.floor(h / 24);
    69→    69→  return `${d}d ago`;
    70→    70→}

    87→    87→// Normalise a raw socket "notification" payload (which may be the AppNotification
    88→    88→// shape OR the leaner {type, orderId, customerName, total, at} shape the realtime
    89→    89→// service emits) into an AppNotification-shaped object.
    90→    90→function normalizeSocketNotif(
    91→    91→  raw: unknown,
    92→    92→  riderKey: string
    93→    93→): AppNotification | null {
    94→    94→  if (!raw || typeof raw !== "object") return null;
    95→    95→  const n = raw as Record<string, unknown>;
    96→    96→  const at = (n.at as string) || (n.createdAt as string) || new Date().toISOString();
    97→    97→  const type = (n.type as string) || "SYSTEM";
    98→    98→  const id =
    99→    99→    (n.id as string) || `socket-${at}-${type}-${(n.orderId as string) || ""}`;
   100→   100→  const title =
   101→   101→    (n.title as string) ||
   102→   102→    (type === "NEW_ORDER" ? "New order 🛒" : type === "DISPATCH" ? "Dispatch alert 🛵" : "New alert");
   103→   103→  const body =
   104→   104→    (n.body as string) ||
   105→   105→    (n.customerName && typeof n.total === "number"
   106→   106→      ? `${n.customerName} · KES ${Number(n.total).toLocaleString("en-KE")}`
   107→   107→      : "");
   108→   108→  return {
   109→   109→    id,
   110→   110→    recipientType: "RIDER",
   111→   111→    recipientId: riderKey,
   112→   112→    type,
   113→   113→    title,
   114→   114→    body,
   115→   115→    read: false,
   116→   116→    orderId: (n.orderId as string) || null,
   117→   117→    createdAt: at,
   118→   118→  };
   119→   119→}
