    68→    68→// ---------- fetch helper ----------
    69→    69→// Auth uses BOTH an httpOnly cookie (for first-party/direct-nav contexts) AND
    70→    70→// a bearer-style `x-admin-token` header (for third-party iframe contexts like
    71→    71→// the preview panel, where SameSite cookies are blocked). The token lives in
    72→    72→// sessionStorage and is attached to every admin fetch.
    73→    73→const TOKEN_KEY = "wb_admin_token";
    74→    74→function getToken(): string | null {
    75→    75→  if (typeof window === "undefined") return null;
    76→    76→  try {
    77→    77→    return sessionStorage.getItem(TOKEN_KEY);
    78→    78→  } catch {
    79→    79→    return null;
    80→    80→  }
    81→    81→}
    82→    82→function setToken(token: string | null) {
    83→    83→  if (typeof window === "undefined") return;
    84→    84→  try {
    85→    85→    if (token) sessionStorage.setItem(TOKEN_KEY, token);
    86→    86→    else sessionStorage.removeItem(TOKEN_KEY);
    87→    87→  } catch {
    88→    88→    // ignore
    89→    89→  }
    90→    90→}
    91→    91→
    92→    92→async function adminFetch<T>(
    93→    93→  url: string,
    94→    94→  init?: RequestInit,
    95→    95→  onUnauthorized?: () => void
    96→    96→): Promise<T> {
    97→    97→  const token = getToken();
    98→    98→  const doFetch = () =>
    99→    99→    fetch(url, {
   100→   100→      ...init,
   101→   101→      headers: {
   102→   102→        "Content-Type": "application/json",
   103→   103→        ...(token ? { "x-admin-token": token } : {}),
   104→   104→        ...(init?.headers ?? {}),
   105→   105→      },
   106→   106→      cache: "no-store",
   107→   107→      credentials: "same-origin",
   108→   108→    });
   109→   109→
   110→   110→  let res = await doFetch();
   111→   111→  if (res.status === 401) {
   112→   112→    // Transient cookie-commit race: retry once after a brief delay.
   113→   113→    await new Promise((r) => setTimeout(r, 500));
   114→   114→    res = await doFetch();
   115→   115→  }
   116→   116→  if (res.status === 401) {
   117→   117→    onUnauthorized?.();
   118→   118→    throw new Error("Unauthorized");
   119→   119→  }
   120→   120→  if (!res.ok) {
   121→   121→    throw new Error(`Request failed (${res.status})`);
   122→   122→  }
   123→   123→  return (await res.json()) as T;
   124→   124→}
