    41→    41→function relativeTime(iso: string): string {
    42→    42→  const then = new Date(iso).getTime();
    43→    43→  if (Number.isNaN(then)) return "";
    44→    44→  const diff = Date.now() - then;
    45→    45→  if (diff < 0) return "just now";
    46→    46→  const s = Math.floor(diff / 1000);
    47→    47→  if (s < 45) return "just now";
    48→    48→  const m = Math.floor(s / 60);
    49→    49→  if (m < 60) return `${m}m ago`;
    50→    50→  const h = Math.floor(m / 60);
    51→    51→  if (h < 24) return `${h}h ago`;
    52→    52→  const d = Math.floor(h / 24);
    53→    53→  if (d < 7) return `${d}d ago`;
    54→    54→  return new Date(iso).toLocaleDateString();
    55→    55→}

    57→    57→function nowHHMM(): string {
    58→    58→  const d = new Date();
    59→    59→  return `${String(d.getHours()).padStart(2, "0")}:${String(
    60→    60→    d.getMinutes()
    61→    61→  ).padStart(2, "0")}`;
    62→    62→}

    64→    64→// True if `now` falls inside the [start, end) duty window. Handles overnight
    65→    65→// windows (e.g. 22:00 → 06:00). Empty schedule ⇒ treat as always on duty.
    66→    66→function isWithinDuty(start: string, end: string, now = nowHHMM()): boolean {
    67→    67→  if (!start || !end) return true;
    68→    68→  if (start === end) return false;
    69→    69→  if (start < end) return now >= start && now < end;
    70→    70→  return now >= start || now < end; // overnight
    71→    71→}

   465→   465→  async function sendChat() {
   466→   466→    const body = chatDraft.trim();
   467→   467→    if (!body || !vendorId || !portal || sending) return;
   468→   468→    const channel = `vendor:${vendorId}`;

   470→   470→      channel,
   471→   471→      senderType: "VENDOR",
   472→   472→      senderId: vendorId,
   473→   473→      senderName: portal.shopName,
   474→   474→      body,
   475→   475→    };
   476→   476→    setSending(true);
   477→   477→    setChatDraft("");
   478→   478→    try {
   479→   479→      const res = await fetch("/api/chat", {
   480→   480→        method: "POST",
   481→   481→        headers: { "Content-Type": "application/json" },
   482→   482→        body: JSON.stringify(payload),
   483→   483→      });
   484→   484→      if (!res.ok) throw new Error(`Failed (${res.status})`);
   485→   485→      const data = await res.json();
   486→   486→      const msg: ChatMessage = data.message;
   487→   487→      setMessages((prev) =>
   488→   488→        prev.some((m) => m.id === msg.id) ? prev : [...prev, msg]
   489→   489→      );
   490→   490→      // also push via the socket so admin support receives it in real time
