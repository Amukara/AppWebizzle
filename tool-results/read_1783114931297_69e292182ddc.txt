    58→function relativeTime(iso: string): string {
    59→  const then = new Date(iso).getTime();
    60→  if (Number.isNaN(then)) return "";
    61→  const diff = Math.max(0, Date.now() - then);
    62→  const s = Math.floor(diff / 1000);
    63→  if (s < 60) return "just now";
    64→  const m = Math.floor(s / 60);
    65→  if (m < 60) return `${m}m ago`;
    66→  const h = Math.floor(m / 60);
    67→  if (h < 24) return `${h}h ago`;
    68→  const d = Math.floor(h / 24);
    69→  return `${d}d ago`;
    70→}

    72→function statusPillClass(status: string): string {
    73→  switch ((status || "").toUpperCase()) {
    74→    case "DELIVERED":
    75→      return "bg-brand-light text-brand";
    76→    case "DISPATCHED":

    80→    case "PLACED":

    82→    default:

    84→  }

    87→// Normalise a raw socket "notification" payload (which may be the AppNotification
    88→// shape OR the leaner {type, orderId, customerName, total, at} shape the realtime
    89→// service emits) into an AppNotification-shaped object.
    90→function normalizeSocketNotif(
    91→  raw: unknown,
    92→  riderKey: string
    93→): AppNotification | null {
    94→  if (!raw || typeof raw !== "object") return null;
    95→  const n = raw as Record<string, unknown>;
    96→  const at = (n.at as string) || (n.createdAt as string) || new Date().toISOString();
    97→  const type = (n.type as string) || "SYSTEM";
    98→  const id =
    99→    (n.id as string) || `socket-${at}-${type}-${(n.orderId as string) || ""}`;
   100→  const title =
   101→    (n.title as string) ||
   102→    (type === "NEW_ORDER" ? "New order 🛒" : type === "DISPATCH" ? "Dispatch alert 🛵" : "New alert");
   103→  const body =
   104→    (n.body as string) ||
   105→    (n.customerName && typeof n.total === "number"
   106→      ? `${n.customerName} · KES ${Number(n.total).toLocaleString("en-KE")}`
   107→      : "");
   108→  return {
   109→    id,
   110→    recipientType: "RIDER",
   111→    recipientId: riderKey,
   112→    type,
   113→    title,
   114→    body,
   115→    read: false,
   116→    orderId: (n.orderId as string) || null,
   117→    createdAt: at,
   118→  };
   119→}

   788→        <Card className="space-y-3 p-4">
