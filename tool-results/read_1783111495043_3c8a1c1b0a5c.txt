    68→// ---------- fetch helper ----------
    69→// Auth uses BOTH an httpOnly cookie (for first-party/direct-nav contexts) AND
    70→// a bearer-style `x-admin-token` header (for third-party iframe contexts like
    71→// the preview panel, where SameSite cookies are blocked). The token lives in
    72→// sessionStorage and is attached to every admin fetch.
    73→const TOKEN_KEY = "wb_admin_token";
    74→function getToken(): string | null {
    75→  if (typeof window === "undefined") return null;
    76→  try {
    77→    return sessionStorage.getItem(TOKEN_KEY);
    78→  } catch {
    79→    return null;
    80→  }
    81→}
    82→function setToken(token: string | null) {
    83→  if (typeof window === "undefined") return;
    84→  try {
    85→    if (token) sessionStorage.setItem(TOKEN_KEY, token);
    86→    else sessionStorage.removeItem(TOKEN_KEY);
    87→  } catch {
    88→    // ignore
    89→  }
    90→}
    91→
    92→async function adminFetch<T>(
    93→  url: string,
    94→  init?: RequestInit,
    95→  onUnauthorized?: () => void
    96→): Promise<T> {
    97→  const token = getToken();
    98→  const doFetch = () =>
    99→    fetch(url, {
   100→      ...init,
   101→      headers: {
   102→        "Content-Type": "application/json",
   103→        ...(token ? { "x-admin-token": token } : {}),
   104→        ...(init?.headers ?? {}),
   105→      },
   106→      cache: "no-store",
   107→      credentials: "same-origin",
   108→    });
   109→
   110→  let res = await doFetch();
   111→  if (res.status === 401) {
   112→    // Transient cookie-commit race: retry once after a brief delay.
   113→    await new Promise((r) => setTimeout(r, 500));
   114→    res = await doFetch();
   115→  }
   116→  if (res.status === 401) {
   117→    onUnauthorized?.();
   118→    throw new Error("Unauthorized");
   119→  }
   120→  if (!res.ok) {
   121→    throw new Error(`Request failed (${res.status})`);
   122→  }
   123→  return (await res.json()) as T;
   124→}

   127→function StatCard({
   128→  label,
   129→  value,
   130→  hint,
   131→  accent,
   132→  icon,
   133→}: {
   134→  label: string;

   139→}) {
   140→  const accentMap: Record<string, string> = {
   141→    brand: "bg-brand-light text-brand",
   142→    gold: "bg-gold/15 text-gold-dark",
   143→    mpesa: "bg-mpesa/10 text-mpesa",
   144→    muted: "bg-muted text-muted-foreground",
   145→  };
   146→  return (
   147→    <Card className="overflow-hidden">
   148→      <CardContent className="p-4">
   149→        <div className="flex items-start justify-between gap-2">
   150→          <div className="min-w-0">
   151→            <p className="text-xs font-medium uppercase tracking-wide text-muted-foreground">
   152→              {label}
   153→            </p>
   154→            <p className="mt-1 truncate text-xl font-bold text-foreground sm:text-2xl">
   155→              {value}
   156→            </p>
   157→            {hint && (
   158→              <p className="mt-1 text-xs text-muted-foreground">{hint}</p>
   159→            )}
   160→          </div>
   161→          {icon && (
   162→            <span
   163→              className={cn(
   164→                "flex size-9 shrink-0 items-center justify-center rounded-lg",
   165→                accentMap[accent]
   166→              )}
   167→            >
   168→              {icon}
   169→            </span>

   171→        </div>
