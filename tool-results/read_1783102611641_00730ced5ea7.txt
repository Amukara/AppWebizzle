    64→// ---------- fetch helper ----------
    65→// Auth uses BOTH an httpOnly cookie (for first-party/direct-nav contexts) AND
    66→// a bearer-style `x-admin-token` header (for third-party iframe contexts like
    67→// the preview panel, where SameSite cookies are blocked). The token lives in
    68→// sessionStorage and is attached to every admin fetch.
    69→const TOKEN_KEY = "wb_admin_token";
    70→function getToken(): string | null {
    71→  if (typeof window === "undefined") return null;
    72→  try {
    73→    return sessionStorage.getItem(TOKEN_KEY);
    74→  } catch {
    75→    return null;
    76→  }
    77→}
    78→function setToken(token: string | null) {
    79→  if (typeof window === "undefined") return;
    80→  try {
    81→    if (token) sessionStorage.setItem(TOKEN_KEY, token);
    82→    else sessionStorage.removeItem(TOKEN_KEY);
    83→  } catch {
    84→    // ignore
    85→  }
    86→}
    87→
    88→async function adminFetch<T>(
    89→  url: string,
    90→  init?: RequestInit,
    91→  onUnauthorized?: () => void
    92→): Promise<T> {
    93→  const token = getToken();
    94→  const doFetch = () =>
    95→    fetch(url, {
    96→      ...init,
    97→      headers: {
    98→        "Content-Type": "application/json",
    99→        ...(token ? { "x-admin-token": token } : {}),
   100→        ...(init?.headers ?? {}),
   101→      },
   102→      cache: "no-store",
   103→      credentials: "same-origin",
   104→    });
   105→
   106→  let res = await doFetch();
   107→  if (res.status === 401) {
   108→    // Transient cookie-commit race: retry once after a brief delay.
   109→    await new Promise((r) => setTimeout(r, 500));
   110→    res = await doFetch();
   111→  }
   112→  if (res.status === 401) {
   113→    onUnauthorized?.();
   114→    throw new Error("Unauthorized");
   115→  }
   116→  if (!res.ok) {
   117→    throw new Error(`Request failed (${res.status})`);
   118→  }
   119→  return (await res.json()) as T;
   120→}

   123→function StatCard({
   124→  label,
   125→  value,
   126→  hint,
   127→  accent,
   128→  icon,

   141→  };

   143→    <Card className="overflow-hidden">
   144→      <CardContent className="p-4">
   145→        <div className="flex items-start justify-between gap-2">
   146→          <div className="min-w-0">
   147→            <p className="text-xs font-medium uppercase tracking-wide text-muted-foreground">
   148→              {label}
   149→            </p>
   150→            <p className="mt-1 truncate text-xl font-bold text-foreground sm:text-2xl">
   151→              {value}
   152→            </p>
   153→            {hint && (
   154→              <p className="mt-1 text-xs text-muted-foreground">{hint}</p>
   155→            )}
   156→          </div>
   157→          {icon && (
   158→            <span
   159→              className={cn(
   160→                "flex size-9 shrink-0 items-center justify-center rounded-lg",
   161→                accentMap[accent]
   162→              )}

   164→              {icon}
